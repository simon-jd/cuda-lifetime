cmake_minimum_required(VERSION 3.20)
project(cuda_lifetime VERSION 0.3.0 LANGUAGES C CXX)

option(CUDA_LIFETIME_ALLOW_UNSUPPORTED_LLVM "Allow local development with LLVM other than 23.1.3 (not release validation)" OFF)
find_package(LLVM REQUIRED CONFIG)
find_package(Clang REQUIRED CONFIG)
if(NOT LLVM_PACKAGE_VERSION VERSION_EQUAL "23.1.3")
  if(NOT CUDA_LIFETIME_ALLOW_UNSUPPORTED_LLVM)
    message(FATAL_ERROR "cuda-lifetime requires LLVM 23.1.3; found ${LLVM_PACKAGE_VERSION}. For development only use -DCUDA_LIFETIME_ALLOW_UNSUPPORTED_LLVM=ON.")
  endif()
  message(WARNING "UNVALIDATED development toolchain: LLVM ${LLVM_PACKAGE_VERSION}, supported version is 23.1.3")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
execute_process(COMMAND "${LLVM_TOOLS_BINARY_DIR}/clang" -print-resource-dir OUTPUT_VARIABLE CUDA_LIFETIME_CLANG_RESOURCE_DIR OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
add_executable(cuda-lifetime-native tool/main.cpp tool/SarifConsumer.cpp checker/AsyncLifetime.cpp checker/Models.cpp)
target_include_directories(cuda-lifetime-native SYSTEM PRIVATE ${LLVM_INCLUDE_DIRS} ${CLANG_INCLUDE_DIRS})
target_include_directories(cuda-lifetime-native PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
separate_arguments(CUDA_LIFETIME_LLVM_DEFINITIONS NATIVE_COMMAND "${LLVM_DEFINITIONS}")
target_compile_options(cuda-lifetime-native PRIVATE ${CUDA_LIFETIME_LLVM_DEFINITIONS})
if(NOT LLVM_ENABLE_RTTI)
  target_compile_options(cuda-lifetime-native PRIVATE -fno-rtti)
endif()
target_compile_definitions(cuda-lifetime-native PRIVATE CUDA_LIFETIME_PRODUCT_VERSION="${PROJECT_VERSION}" CUDA_LIFETIME_LLVM_VERSION="${LLVM_PACKAGE_VERSION}" CUDA_LIFETIME_CLANG_RESOURCE_DIR="${CUDA_LIFETIME_CLANG_RESOURCE_DIR}")
if(TARGET clang-cpp)
  target_link_libraries(cuda-lifetime-native PRIVATE clang-cpp)
  if(TARGET LLVM)
    target_link_libraries(cuda-lifetime-native PRIVATE LLVM)
  endif()
else()
  target_link_libraries(cuda-lifetime-native PRIVATE clangTooling clangStaticAnalyzerFrontend clangStaticAnalyzerCheckers clangStaticAnalyzerCore clangFrontend clangSerialization clangDriver clangParse clangSema clangAnalysis clangASTMatchers clangAST clangLex clangBasic)
endif()
install(TARGETS cuda-lifetime-native RUNTIME DESTINATION bin)
include(CTest)
find_package(Python3 COMPONENTS Interpreter)
if(BUILD_TESTING AND Python3_Interpreter_FOUND)
  add_test(NAME seed-regression COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/run_seed.py --native $<TARGET_FILE:cuda-lifetime-native> --out ${CMAKE_CURRENT_BINARY_DIR}/seed-regression)
  add_test(NAME native-regression COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/native/run.py $<TARGET_FILE:cuda-lifetime-native>)
endif()
